
const D = window.__D__, $ = s => document.querySelector(s);
const esc = s => String(s).replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
const PAGES = {principles:"Principles worth carrying", dead:"What didn't survive", about:"About this archive"};

let sec = {kind:'home'};   // home | topic:<cat> | page:<id>
let cur = null;            // entry index or null
let q = '';

const tierLabel = t => t==='A' ? 'fully read' : 'summarized';
const bars = rows => {
  const max = Math.max(...rows.map(r => r[1]));
  return `<div class="chart minichart">${rows.map(([lbl,n,alt]) =>
    `<div class="bar"><span class="lbl">${esc(lbl)}</span>` +
    `<span class="track"><span class="fill${alt?' alt':''}" style="width:${Math.round(n/max*100)}%"></span></span>` +
    `<span class="v">${n}</span></div>`).join('')}</div>`;
};
const found = () => q ? D.e.filter(e => e.q.includes(q) || e.sum.toLowerCase().includes(q)) : null;
const inTopic = c => (found() || D.e).filter(e => e.cat.includes(c));

/* ---------------- column 1 ---------------- */
function renderNav(){
  const hits = found();
  let h = '<div class="navsec"><h2>Archive</h2>';
  for(const c of D.cats){
    const n = (hits || D.e).filter(e=>e.cat.includes(c)).length;
    if(hits && !n) continue;
    h += `<button class="navitem" data-cat="${esc(c)}" aria-current="${sec.kind==='topic'&&sec.cat===c}">
      <span>${esc(c)}</span><span class="n">${n}</span></button>`;
  }
  h += '</div><div class="navsec"><h2>Collection</h2>';
  h += `<button class="navitem" data-page="principles" aria-current="${sec.kind==='page'&&sec.id==='principles'}"><span>Principles worth carrying</span><span class="n">${D.p.reduce((n,g)=>n+g.items.length,0)}</span></button>`;
  h += `<button class="navitem" data-page="dead" aria-current="${sec.kind==='page'&&sec.id==='dead'}"><span>What didn't survive</span><span class="n">${D.d.length}</span></button>`;
  h += `<button class="navitem" data-page="about" aria-current="${sec.kind==='page'&&sec.id==='about'}"><span>About this archive</span></button>`;
  h += '</div>';
  $('#nav').innerHTML = h;
}

/* ---------------- column 2 ---------------- */
function renderList(){
  const hits = found();
  let path, title, blurb, items;

  if(hits && sec.kind!=='page'){
    path='/search'; title=`${hits.length} ${hits.length===1?'result':'results'}`;
    blurb=`for “${esc(q)}”`; items=hits;
  } else if(sec.kind==='topic'){
    path='/archive'; title=sec.cat; blurb=D.blurb[sec.cat]||''; items=inTopic(sec.cat);
  } else if(sec.kind==='page'){
    path='/collection'; title=PAGES[sec.id]; blurb=''; items=[];
  } else {
    path='/'; title='Attic'; blurb='Pick a topic to begin, or search.'; items=[];
  }

  $('#listhead').innerHTML =
    `<p class="path">${esc(path)}</p><h2>${esc(title)}</h2>${blurb?`<p>${blurb}</p>`:''}`;

  if(!items.length){
    $('#listscroll').innerHTML = sec.kind==='home'
      ? `<p class="listempty">Ten topics, ${D.e.length} entries.</p>`
      : (sec.kind==='page' ? '' : `<p class="listempty">Nothing matches that.</p>`);
    return;
  }
  $('#listscroll').innerHTML = items.map(e=>{
    const meta = [e.a || e.s, e.y].filter(Boolean).join(' · ');
    return `<button class="li" data-e="${e.i}" aria-current="${cur===e.i}">
      ${esc(e.t)}<span class="sub">${esc(meta)}</span></button>`;
  }).join('');
}

/* ---------------- column 3 ---------------- */
function entryHTML(e){
  const sib = inTopic(sec.kind==='topic' ? sec.cat : e.cat[0]);
  const k = sib.findIndex(x=>x.i===e.i);
  const prev = sib[k-1], next = sib[k+1];
  const bits = [e.a,e.s,e.y].filter(Boolean).map(esc);
  return `<button class="backbtn" data-back="1">← All entries</button>
  <p class="crumb">${esc(e.cat[0])}</p>
  <h2 class="page">${esc(e.t)}</h2>
  <p class="byline">${bits.map(b=>`<span>${b}</span>`).join('')}<span class="t">${tierLabel(e.tier)}</span></p>
  <div class="prose">${e.sum}</div>
  ${e.kp.length?`<div class="cardblock kp"><span class="lab acc">Key points</span><ol>${e.kp.map(x=>`<li><span>${esc(x)}</span></li>`).join('')}</ol></div>`:''}
  ${e.sr?`<div class="cardblock sr"><span class="lab">Still relevant?</span>${e.sr}</div>`:''}
  <div class="srcs"><h4>Source${e.links.length>1?'s':''}</h4>
    ${e.links.map(l=>`<a href="${esc(l.u)}" target="_blank" rel="noopener noreferrer">${esc(l.u)}</a>`).join('')}</div>
  <div class="tagrow">${e.cat.map(c=>`<button class="tag" data-cat="${esc(c)}">${esc(c)}</button>`).join('')}</div>
  <nav class="navback">
    ${prev?`<button data-e="${prev.i}"><span>Previous</span>${esc(prev.t)}</button>`:'<span></span>'}
    ${next?`<button data-e="${next.i}" style="text-align:right;margin-left:auto"><span>Next</span>${esc(next.t)}</button>`:''}
  </nav>`;
}

function topicIntro(c){
  const items = inTopic(c);
  return `<p class="crumb">Archive</p><h2 class="page">${esc(c)}</h2>
    <p class="pagesub">${esc(D.blurb[c]||'')}</p>
    <p>${items.length} ${items.length===1?'entry':'entries'}. Choose one from the list to read it.</p>`;
}

    <p>This is an attempt to make good on it. <strong>286 saved links</strong>, opened again one at a time: what each piece argued, what still holds, what has been overtaken, and what is simply gone. They are gathered here as ${D.e.length} entries, because the course often assigned several pieces making the same point.</p>
    <p class="small">Pick a topic on the left, or search.</p>
  </div>
  <div class="chart"><h4>What was in it</h4>
    ${counts.map(x=>`<div class="bar"><span class="lbl">${esc(x.c)}</span>

/* ---------------- render ---------------- */
function render(){
  renderNav(); renderList();
  let h;
  if(cur !== null){ const e = D.e.find(x=>x.i===cur); h = e ? entryHTML(e) : home(); }
  else if(sec.kind==='topic') h = topicIntro(sec.cat);
  else if(sec.kind==='page') h = pageHTML[sec.id]();
  else h = home();
  $('#main').innerHTML = h;
  const showList = (sec.kind==='topic') || (q && sec.kind!=='page');
  document.body.dataset.list = showList ? 'on' : 'off';
  document.body.dataset.mode = cur!==null ? 'detail' : 'browse';
  $('#scroller').scrollTop = 0; window.scrollTo(0,0);
  document.body.dataset.nav = '';
}

/* ---------------- deep links ---------------- */
// A published artifact only passes through a bare #token (letters, digits and
// . _ ~ -). It never passes #key=value and never a query string, so the whole
// state is encoded as ONE key: an entry id, a topic slug, or a page id.
// The build guarantees those 131 keys are unique and hash-safe.
const PAGE_KEYS = Object.keys(PAGES);
const catForSlug = s => D.cats.find(c => D.slug[c] === s);

function hashFor(){
  if(cur !== null){ const e = D.e.find(x => x.i === cur); if(e) return e.k; }
  if(sec.kind === 'topic') return D.slug[sec.cat] || '';
  if(sec.kind === 'page')  return sec.id;
  return '';
}

function applyHash(){
  const h = (location.hash || '').replace(/^#/, '').trim().toLowerCase();
  q = ''; const qi = $('#q'); if(qi) qi.value = '';
  if(!h){ sec = {kind:'home'}; cur = null; return; }
  if(PAGE_KEYS.includes(h)){ sec = {kind:'page', id:h}; cur = null; return; }
  const cat = catForSlug(h);
  if(cat){ sec = {kind:'topic', cat}; cur = null; return; }
  const e = D.e.find(x => x.k === h);
  if(e){ sec = {kind:'topic', cat:e.cat[0]}; cur = e.i; return; }
  sec = {kind:'home'}; cur = null;          // unknown key: fail to the front page
}

let navLock = false;
function syncURL(replace){
  const h = hashFor();
  if((location.hash || '').replace(/^#/, '') === h) return;
  const url = h ? '#' + h : location.pathname + location.search;
  navLock = true;
  try{
    history[replace ? 'replaceState' : 'pushState'](null, '', url);
  }catch(err){
    location.hash = h;        // some embedded frames refuse history writes
  }
  setTimeout(() => { navLock = false; }, 0);
}

function go(){ render(); syncURL(false); }
function onRoute(){ if(navLock) return; applyHash(); render(); }
addEventListener('popstate', onRoute);
addEventListener('hashchange', onRoute);

document.addEventListener('click', ev=>{
  const t = ev.target.closest('[data-e],[data-cat],[data-page],[data-back],.brand,#menubtn,.scrim');
  if(!t) return;
  if(t.classList.contains('scrim')||t.id==='menubtn'){
    document.body.dataset.nav = document.body.dataset.nav==='open'?'':'open'; return;
  }
  if(t.classList.contains('brand')){ sec={kind:'home'}; cur=null; q=''; $('#q').value=''; return go(); }
  if(t.dataset.back){ cur=null; return go(); }
  if(t.dataset.page){ sec={kind:'page',id:t.dataset.page}; cur=null; return go(); }
  if(t.dataset.e!==undefined){
    const e = D.e.find(x=>x.i===+t.dataset.e);
    if(sec.kind!=='topic' || !e.cat.includes(sec.cat)){
      if(!q) sec={kind:'topic',cat:e.cat[0]};
    }
    cur = e.i; return go();
  }
  if(t.dataset.cat){
    sec={kind:'topic',cat:t.dataset.cat};
    const first = inTopic(sec.cat)[0];
    cur = first ? first.i : null;
    return go();
  }
});

$('#q').addEventListener('input', e=>{
  q = e.target.value.trim().toLowerCase();
  if(q && sec.kind==='page') sec={kind:'home'};
  const hits = found();
  cur = (q && hits && hits.length) ? hits[0].i : null;
  render(); syncURL(true);          // replace, so a search is not 20 history entries
});

// --- footer controls: accent colour, light / dark ---
const R = document.documentElement, MQ = matchMedia('(prefers-color-scheme: dark)');

// light / dark: follow the system until the reader chooses, then keep their choice
let modeLocked = false;
function setMode(m){
  R.dataset.mode = m;
  const b = $('#modetog');
  if(b) b.setAttribute('aria-label', m==='dark' ? 'Switch to light mode' : 'Switch to dark mode');
}
try{ const s = localStorage.getItem('mode'); if(s==='dark'||s==='light'){ modeLocked = true; setMode(s); } }catch(e){}
if(!modeLocked) setMode(MQ.matches ? 'dark' : 'light');
MQ.addEventListener('change', e => { if(!modeLocked) setMode(e.matches ? 'dark' : 'light'); });
$('#modetog').addEventListener('click', () => {
  const next = R.dataset.mode==='dark' ? 'light' : 'dark';
  modeLocked = true; setMode(next);
  try{ localStorage.setItem('mode', next); }catch(e){}
});

// accent colour
const PALS = ['madder','plum'];
function setPal(p){ if(PALS.includes(p)) R.dataset.pal = p; }
try{ const s = localStorage.getItem('pal'); if(s) setPal(s); }catch(e){}
$('#paltog').addEventListener('click', () => {
  const next = PALS[(PALS.indexOf(R.dataset.pal) + 1) % PALS.length];
  setPal(next);
  try{ localStorage.setItem('pal', next); }catch(e){}
});

applyHash();
render();
syncURL(true);
